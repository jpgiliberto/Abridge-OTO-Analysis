APP <- c("ID563","ID606","ID888","ID006")

LMunSchData <- SignalRaw |>
  filter(Metric == "Time On Unscheduled Days")|>
  filter(GroupData !="Transition") |>
  filter(StudyID %in% APP)|> 
  group_by(StudyID,StartDate,GroupData) |> 
  summarise(Unsch = sum(Numerator,na.rm =T)) |> 
   left_join(KeyDenominator, join_by(StudyID,StartDate)) |> 
# left_join(KeyUserHigh) |> 
 # filter(HighUser) |> 
  mutate(HrsPerDay = Unsch*20/`per Appointment`/60) |> 
  select(StudyID, Unsch, GroupData, HrsPerDay)

LMunSch <- lmer(formula = HrsPerDay ~ (GroupData) + (1|StudyID), data = LMunSchData)
